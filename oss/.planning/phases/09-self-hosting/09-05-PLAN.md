**kv-adapter-wrapper.ts** -- Wraps a KvStoreDO stub to implement KeyValueStore:
```typescript
import type { KeyValueStore } from '../interfaces'

/**
 * Wraps a KvStoreDO Durable Object stub to provide
 * the KeyValueStore interface to the gateway.
 */
export class SelfHostedKvAdapter implements KeyValueStore {
  constructor(private stub: any) {} // DO stub from env binding

  async get(key: string): Promise<string | null>
  async get<T>(key: string, type: 'json'): Promise<T | null>
  async get(key: string, type?: string): Promise<unknown> {
    if (type === 'json') {
      return this.stub.kvGetJson(key)
    }
    return this.stub.kvGet(key)
  }

  async put(key: string, value: string): Promise<void> {
    return this.stub.kvPut(key, value)
  }

  async delete(key: string): Promise<void> {
    return this.stub.kvDelete(key)
  }

  async list(options?: { prefix?: string }): Promise<{ keys: Array<{ name: string }> }> {
    return this.stub.kvList(options?.prefix)
  }
}
```

**db-adapter-wrapper.ts** -- Wraps a UsageDbDO stub to implement UsageDatabase:
```typescript
import type { UsageDatabase, PreparedStatement, BoundStatement } from '../interfaces'

/**
 * Wraps a UsageDbDO Durable Object stub to provide
 * the UsageDatabase interface (D1-compatible) to the gateway.
 */
export class SelfHostedDbAdapter implements UsageDatabase {
  constructor(private stub: any) {}

  prepare(sql: string): PreparedStatement {
    return new SelfHostedPreparedStatement(this.stub, sql)
  }
}

class SelfHostedPreparedStatement implements PreparedStatement {
  constructor(private stub: any, private sql: string) {}

  bind(...values: unknown[]): BoundStatement {
    return new SelfHostedBoundStatement(this.stub, this.sql, values)
  }

  async all<T>(): Promise<{ results: T[] }> {
    return this.stub.query(this.sql)
  }

  async first<T>(): Promise<T | null> {
    return this.stub.queryFirst(this.sql)
  }

  async run(): Promise<{ meta: { changes?: number } }> {
    return this.stub.execute(this.sql)
  }
}

class SelfHostedBoundStatement implements BoundStatement {
  constructor(private stub: any, private sql: string, private values: unknown[]) {}

  async all<T>(): Promise<{ results: T[] }> {
    return this.stub.query(this.sql, ...this.values)
  }

  async first<T>(): Promise<T | null> {
    return this.stub.queryFirst(this.sql, ...this.values)
  }

  async run(): Promise<{ meta: { changes?: number } }> {
    return this.stub.execute(this.sql, ...this.values)
  }
}
```

**self-hosted-entry.ts** -- A separate Worker entry point for self-hosted mode. This file:
1. Imports the Hono app
2. On each request, detects runtime = 'self-hosted' from the RUNTIME binding
3. Creates self-hosted adapters (KV from KvStoreDO, UsageDB from UsageDbDO, rate limiters from InMemoryRateLimiter)
4. Passes the adapted env to the Hono app
5. Exports KvStoreDO, UsageDbDO, and TokenCoordinator DO classes

```typescript
import app from './app'
import { handleScheduled } from './scheduled'
import { SelfHostedKvAdapter } from './runtime/self-hosted/kv-adapter-wrapper'
import { SelfHostedDbAdapter } from './runtime/self-hosted/db-adapter-wrapper'
import { InMemoryRateLimiter } from './runtime/self-hosted/rate-limiter'
import { loadConfigFromObject, DEFAULT_CONFIG } from './runtime/config'

export { KvStoreDO } from './runtime/self-hosted/kv-store-do'
export { UsageDbDO } from './runtime/self-hosted/usage-db-do'
export { TokenCoordinator } from './durable-objects/token-coordinator'

// Singleton rate limiters (persist across requests in the same worker instance)
let rateLimiters: Record<string, InMemoryRateLimiter> | null = null

function getRateLimiters(config: typeof DEFAULT_CONFIG) {
  if (!rateLimiters) {
    rateLimiters = {
      free: new InMemoryRateLimiter(config.rate_limits.free.requests_per_minute, 60),
      pro: new InMemoryRateLimiter(config.rate_limits.pro.requests_per_minute, 60),
      enterprise: new InMemoryRateLimiter(config.rate_limits.enterprise.requests_per_minute, 60),
      ip: new InMemoryRateLimiter(config.rate_limits.ip.requests_per_10s, 10),
    }
  }
  return rateLimiters
}

export default {
  async fetch(request: Request, env: Record<string, any>, ctx: ExecutionContext): Promise<Response> {
    const config = DEFAULT_CONFIG // Config loaded from defaults; overrides via bindings

    // Get DO stubs for KV and Usage DB
    const kvId = env.KV_STORE.idFromName('default')
    const kvStub = env.KV_STORE.get(kvId)
    const usageId = env.USAGE_DB_DO.idFromName('default')
    const usageStub = env.USAGE_DB_DO.get(usageId)

    const limiters = getRateLimiters(config)

    // Build adapted environment
    const adaptedEnv = {
      ...env,
      AUTH_KV: new SelfHostedKvAdapter(kvStub),
      USAGE_DB: new SelfHostedDbAdapter(usageStub),
      RATE_LIMIT_FREE: limiters.free,
      RATE_LIMIT_PRO: limiters.pro,
      RATE_LIMIT_ENTERPRISE: limiters.enterprise,
      RATE_LIMIT_IP: limiters.ip,
      TOKEN_COORDINATOR: {
        getStub() {
          const id = env.TOKEN_COORDINATOR.idFromName('default')
          return env.TOKEN_COORDINATOR.get(id)
        }
      },
      ENVIRONMENT: 'self-hosted',
      FEELR_CONFIG: {
        runtime: 'self-hosted' as const,
        billing: config.billing,
        encryption: config.encryption,
      },
    }

    return app.fetch(request, adaptedEnv, ctx)
  },

  // Scheduled handler for retention cleanup
  // In self-hosted mode, triggered via /__scheduled endpoint or setInterval
  async scheduled(event: ScheduledEvent, env: Record<string, any>, ctx: ExecutionContext): Promise<void> {
    // Build adapted USAGE_DB for scheduled handler
    const usageId = env.USAGE_DB_DO.idFromName('default')
    const usageStub = env.USAGE_DB_DO.get(usageId)

    const adaptedEnv = {
      ...env,
      USAGE_DB: new SelfHostedDbAdapter(usageStub),
    }

    return handleScheduled(event, adaptedEnv as any, ctx)
  },
}
```

Key design points:
- Rate limiters are singletons (created once per worker instance, survive across requests)
- DO stubs are created per-request (they're cheap -- just an ID lookup)
- TokenCoordinatorClient adapter is inline (simple getStub() wrapper)
- The existing cloud entry point (index.ts) is NOT modified -- it continues to work for Cloudflare deployment

The `dashboard-static` disk service serves the dashboard's static export files. workerd can serve static files from a disk directory when a request path matches a file. Non-API requests that don't match the gateway worker routes can fall through to the static file server. However, workerd's socket routing sends ALL requests to a single service. So the gateway worker itself needs to check if a request is for the API or the dashboard, and proxy/serve accordingly.

REVISED: Instead of a separate disk service, the gateway worker should serve dashboard static files for non-API paths. Add a simple static file serving handler in self-hosted-entry.ts that reads from the dashboard directory. OR, use the approach from the research: workerd's `disk` service can be referenced as an internal service binding. The gateway worker can subrequest to it.

SIMPLEST approach: Use a lightweight Node.js static server for the dashboard as a second process under s6-overlay. The gateway worker handles all API routes (port 8080 internally), and a simple static server handles dashboard routes (port 3000 internally). An internal nginx or the s6 routing handles path-based dispatch. BUT the user decided on single port (8080).

BEST approach: Add a catch-all route in self-hosted-entry.ts that serves static files from a `DASHBOARD` service binding (a disk service in capnp). Update capnp to bind the dashboard disk as a service binding:
```capnp
  bindings = [
    ...existing...
    (name = "DASHBOARD", service = "dashboard"),
  ],
```
Then in self-hosted-entry.ts, after the Hono app returns 404, check if the request path maps to a dashboard static file and subrequest to the DASHBOARD service.

Actually, the simpler approach: In the Hono app's 404 handler (or a catch-all middleware), for self-hosted mode, subrequest to the dashboard service. But this means modifying app.ts which should stay runtime-agnostic.

FINAL approach: In self-hosted-entry.ts's fetch handler, try the Hono app first. If it returns 404, subrequest to the DASHBOARD service binding for static file serving. This keeps app.ts clean.

```typescript
// In self-hosted-entry.ts fetch:
const response = await app.fetch(request, adaptedEnv, ctx)
if (response.status === 404 && env.DASHBOARD) {
  // Try serving as static file from dashboard
  const dashboardResponse = await env.DASHBOARD.fetch(request)
  if (dashboardResponse.status !== 404) {
    return dashboardResponse
  }
  // Also try with /index.html for SPA routing
  const url = new URL(request.url)
  if (!url.pathname.includes('.')) {
    const indexRequest = new Request(new URL('/index.html', request.url), request)
    const indexResponse = await env.DASHBOARD.fetch(indexRequest)
    if (indexResponse.status !== 404) {
      return indexResponse
    }
  }
}
return response
```

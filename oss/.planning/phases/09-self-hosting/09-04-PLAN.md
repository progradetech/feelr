**Before:**

```typescript
import type { Env } from 'hono'
import type { ApiKeyRecord } from '../auth/types'

export interface RateLimitBinding {
  limit(options: { key: string }): Promise<{ success: boolean }>
}

export interface AppEnv extends Env {
  Bindings: {
    ENVIRONMENT: string
    AUTH_KV: KVNamespace
    TOKEN_COORDINATOR: DurableObjectNamespace
    ENCRYPTION_KEY: string
    ADMIN_TOKEN: string
    SLACK_CLIENT_ID: string
    SLACK_CLIENT_SECRET: string

  }
  Variables: { ... }
}
```

**After:**
```typescript
import type { Env } from 'hono'
import type { ApiKeyRecord } from '../auth/types'
import type {
  KeyValueStore,
  UsageDatabase,
  RateLimiter,
  TokenCoordinatorClient,
} from '../runtime/interfaces'

export interface AppEnv extends Env {
  Bindings: {
    ENVIRONMENT: string
    AUTH_KV: KeyValueStore
    TOKEN_COORDINATOR: TokenCoordinatorClient
    ENCRYPTION_KEY: string
    ADMIN_TOKEN: string
    SLACK_CLIENT_ID: string
    SLACK_CLIENT_SECRET: string
    USAGE_DB: UsageDatabase
    RATE_LIMIT_FREE: RateLimiter
    RATE_LIMIT_PRO: RateLimiter
    RATE_LIMIT_ENTERPRISE: RateLimiter
    RATE_LIMIT_IP: RateLimiter
    /** Runtime config loaded from feelr.yaml (self-hosted) or defaults (cloud) */
    FEELR_CONFIG?: import('../runtime/interfaces').FeelrConfig
  }
  Variables: {
    requestId: string
    apiKey: string | null
    apiKeyRecord?: ApiKeyRecord
  }
}
```

**Key changes:**
- Remove the local `RateLimitBinding` interface (replaced by `RateLimiter` from interfaces.ts)
- Replace `KVNamespace` with `KeyValueStore`
- Replace `D1Database` with `UsageDatabase`
- Replace `DurableObjectNamespace` with `TokenCoordinatorClient`
- Replace all `RateLimitBinding` references with `RateLimiter`
- Add optional `FEELR_CONFIG` binding for runtime config access
- Keep `Variables` section unchanged
- Keep all string bindings (ENCRYPTION_KEY, ADMIN_TOKEN, etc.) unchanged

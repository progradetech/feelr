```go
// DryRunResult holds the dry-run analysis output.
type DryRunResult struct {
    Valid       bool
    Errors      []string
    Steps       []DryRunStep
}

type DryRunStep struct {
    ID             string
    Connector      string
    Action         string
    ResolvedParams map[string]string
    Condition      string
    ConditionResult string // "true", "false", "skipped (no condition)"
    WouldSkip      bool
}

// DryRun validates a chain definition and simulates execution with mock data.
// It resolves all template expressions against mock step outputs and provided params,
// showing what the execution plan would look like.
func DryRun(chain *Chain, params map[string]string, w io.Writer) (*DryRunResult, error)
```

**DryRun implementation:**
1. Validate chain via ValidateChain. If invalid, return errors immediately.
2. Check required params are provided. Report missing params as errors.
3. Build mock ResolveContext: params from input, steps initially empty.
4. For each step:
   a. Parse uses -> connector, action
   b. Generate mock output via MockStepOutput(connector, action)
   c. Store mock output in ctx.Steps[step.ID]
   d. Resolve all with values using ResolveTemplate against mock context
   e. Evaluate condition if present (against mock data)
   f. Record DryRunStep with resolved params and condition result
5. Print execution plan to writer:

```go
func ResolveChain(nameOrPath string) (*Chain, error) {
    // 1. Direct file path (has extension)
    ext := filepath.Ext(nameOrPath)
    if ext == ".yaml" || ext == ".yml" || ext == ".json" {
        return LoadChain(nameOrPath)
    }

    // 2. Search locations for chain name
    searchPaths := []string{}

    // a. Current directory
    searchPaths = append(searchPaths,
        nameOrPath+".yaml",
        nameOrPath+".yml",
        nameOrPath+".json",
    )

    // b. chains/ subdirectory (relative to cwd)
    searchPaths = append(searchPaths,
        filepath.Join("chains", nameOrPath+".yaml"),
        filepath.Join("chains", nameOrPath+".yml"),
        filepath.Join("chains", nameOrPath+".json"),
    )

    // c. Built-in chains directory (relative to executable)
    if execPath, err := os.Executable(); err == nil {
        execDir := filepath.Dir(execPath)
        searchPaths = append(searchPaths,
            filepath.Join(execDir, "chains", nameOrPath+".yaml"),
            filepath.Join(execDir, "chains", nameOrPath+".yml"),
        )
        // Also check parent of exec dir (for dev: cli/../chains/)
        searchPaths = append(searchPaths,
            filepath.Join(execDir, "..", "chains", nameOrPath+".yaml"),
        )
    }

    for _, p := range searchPaths {
        if _, err := os.Stat(p); err == nil {
            return LoadChain(p)
        }
    }

    return nil, fmt.Errorf("chain %q not found. Use \"feelr chain list\" to see available chains", nameOrPath)
}
```

Create `apps/gateway/src/middleware/rate-limiter.ts` with two exported middleware functions:

1. `ipRateLimiter` -- IP-based pre-auth defense:
   ```ts
   import { createMiddleware } from 'hono/factory'
   import type { AppEnv } from '../lib/types'
   import { FeelrError } from '../lib/errors'

   /**
    * IP-based rate limiter -- runs BEFORE API key auth.
    * Provides defense against brute-force auth attempts.
    * Uses Cloudflare native Rate Limiting binding (100 req/10s per IP).
    */
   export const ipRateLimiter = createMiddleware<AppEnv>(async (c, next) => {
     const ip = c.req.header('cf-connecting-ip') ?? c.req.header('x-forwarded-for') ?? 'unknown'
     const result = await c.env.RATE_LIMIT_IP.limit({ key: ip })
     if (!result.success) {
       throw new FeelrError('RATE_LIMITED', {
         message: 'Too many requests from this IP. Please slow down.',
         hint: 'retry',
         status: 429,
       })
     }
     await next()
   })
   ```

2. `keyRateLimiter` -- per-key tier-based rate limiter (runs AFTER api-key middleware):
   ```ts
   /**
    * Per-key tier-based rate limiter -- runs AFTER API key auth.
    * Checks the key's tier and uses the corresponding RateLimit binding.
    * On 429, throws RATE_LIMITED with Retry-After derived from a 60s window.
    *
    * Rate limit bindings are per-location (Cloudflare edge) and use
    * fixed-window with permissive eventual consistency -- ideal for
    * bursty agent traffic patterns.
    */
   export const keyRateLimiter = createMiddleware<AppEnv>(async (c, next) => {
     const record = c.get('apiKeyRecord')
     if (!record) {
       // No API key record means auth middleware hasn't run or failed.
       // Skip rate limiting -- auth errors take priority.
       await next()
       return
     }

     const tier = record.tier ?? 'free'
     const bindingMap: Record<string, typeof c.env.RATE_LIMIT_FREE> = {
       free: c.env.RATE_LIMIT_FREE,
       pro: c.env.RATE_LIMIT_PRO,
       enterprise: c.env.RATE_LIMIT_ENTERPRISE,
     }
     const binding = bindingMap[tier] ?? c.env.RATE_LIMIT_FREE

     const result = await binding.limit({ key: record.shortToken })
     if (!result.success) {
       // Estimate Retry-After as remaining seconds in the 60s window.
       // Since native binding is fixed-window, worst case is 60s.
       const retryAfter = 60
       c.header('Retry-After', String(retryAfter))
       throw new FeelrError('RATE_LIMITED', {
         message: `Rate limit exceeded. Try again in ${retryAfter} seconds.`,
         hint: 'retry',
         status: 429,
       })
     }

     await next()
   })
   ```

Important implementation notes:
- Import `RateLimitTier` and `TIER_LIMITS` from `../auth/types` for type safety.
- The `record.tier ?? 'free'` handles backward compatibility for existing keys created before Phase 7 that don't have a tier field.
- The IP rate limiter uses `cf-connecting-ip` (Cloudflare's real IP header) with fallback to `x-forwarded-for`, then `'unknown'`.
- The Retry-After value of 60 is conservative (full window period). The native binding doesn't expose time-until-reset.

1. Create `apps/gateway/src/middleware/rate-limit-headers.ts`:
   ```ts
   import { createMiddleware } from 'hono/factory'
   import type { AppEnv } from '../lib/types'
   import { TIER_LIMITS } from '../auth/types'
   import type { RateLimitTier } from '../auth/types'

   /**
    * Approximate in-memory request counter for RateLimit-Remaining header.
    * Resets on Worker restart -- this is intentionally approximate.
    * Key: `${shortToken}:${minuteBucket}` -> count
    */
   const requestCounts = new Map<string, number>()

   /**
    * Get the current minute bucket string for approximate counting.
    */
   function getMinuteBucket(): string {
     const now = new Date()
     return `${now.getUTCFullYear()}-${now.getUTCMonth()}-${now.getUTCDate()}-${now.getUTCHours()}-${now.getUTCMinutes()}`
   }

   /**
    * Rate limit headers middleware.
    * Adds RateLimit-Limit, RateLimit-Remaining, and RateLimit-Reset headers
    * to every /v1/* response. These headers use the draft RateLimit header
    * fields (IETF draft-ietf-httpapi-ratelimit-headers).
    *
    * RateLimit-Remaining is approximate (in-memory counter, resets on Worker restart).
    * RateLimit-Reset is the number of seconds until the current window resets.
    */
   export const rateLimitHeaders = createMiddleware<AppEnv>(async (c, next) => {
     await next()

     // Only add headers if we have an authenticated key
     const record = c.get('apiKeyRecord')
     if (!record) return

     const tier: RateLimitTier = (record.tier as RateLimitTier) ?? 'free'
     const limit = TIER_LIMITS[tier] ?? TIER_LIMITS.free

     // Approximate remaining count
     const bucket = getMinuteBucket()
     const counterKey = `${record.shortToken}:${bucket}`
     const currentCount = (requestCounts.get(counterKey) ?? 0) + 1
     requestCounts.set(counterKey, currentCount)

     // Clean up old buckets (keep only current bucket per key)
     // This is a simple memory management strategy
     for (const [key] of requestCounts) {
       if (!key.endsWith(`:${bucket}`)) {
         requestCounts.delete(key)
       }
     }

     const remaining = Math.max(0, limit - currentCount)

     // Calculate seconds until next minute boundary
     const now = new Date()
     const secondsUntilReset = 60 - now.getUTCSeconds()

     c.header('RateLimit-Limit', String(limit))
     c.header('RateLimit-Remaining', String(remaining))
     c.header('RateLimit-Reset', String(secondsUntilReset))
   })
   ```

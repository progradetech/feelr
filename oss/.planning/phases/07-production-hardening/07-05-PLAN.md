1. In `apps/dashboard/src/lib/hooks/use-usage.ts`, add a new `useRateLimits` hook after the existing hooks:

   Add the type:
   ```ts
   export interface RateLimitInfo {
     short_token: string
     label: string | null
     tier: string
     limit: number
     usage_last_minute: number
     throttle_count_24h: number
   }
   ```

   Add the hook:
   ```ts
   /**
    * Fetches per-key rate limit information from /internal/rate-limits.
    * Refreshes every 30 seconds to show near-real-time usage.
    */
   export function useRateLimits() {
     return useSWR<RateLimitInfo[]>(
       'admin-rate-limits',
       async () => {
         const res = await gatewayFetch('/internal/rate-limits')
         const json = await res.json()
         return json.data ?? []
       },
       { refreshInterval: 30000 }
     )
   }
   ```

   Make sure `useSWR` and `gatewayFetch` are already imported at the top of the file (they should be from Phase 6).

   ```tsx
   {/* Rate Limit Status */}
   {rateLimits && rateLimits.length > 0 && (
     <div className="space-y-3">
       <h2 className="text-sm font-medium text-zinc-300">Rate Limits</h2>
       <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
         {rateLimits.map((rl: RateLimitInfo) => {
           const usagePercent = rl.limit > 0 ? Math.min(100, Math.round((rl.usage_last_minute / rl.limit) * 100)) : 0;
           const isNearLimit = usagePercent >= 80;
           const isAtLimit = usagePercent >= 100;
           return (
             <div
               key={rl.short_token}
               className="rounded-lg border border-zinc-800 bg-zinc-900/50 p-4"
             >
               <div className="flex items-center justify-between">
                 <span className="text-sm font-medium text-zinc-200 truncate">
                   {rl.label || `fk_...${rl.short_token}`}
                 </span>
                 <span className={`rounded-full px-2 py-0.5 text-xs font-medium ${
                   rl.tier === 'enterprise'
                     ? 'bg-purple-900/50 text-purple-300'
                     : rl.tier === 'pro'
                     ? 'bg-blue-900/50 text-blue-300'
                     : 'bg-zinc-800 text-zinc-400'
                 }`}>
                   {rl.tier}
                 </span>
               </div>
               {/* Usage bar */}
               <div className="mt-3">
                 <div className="flex items-center justify-between text-xs">
                   <span className="text-zinc-400">
                     {rl.usage_last_minute} / {rl.limit} req/min
                   </span>
                   <span className={
                     isAtLimit ? 'text-red-400' : isNearLimit ? 'text-amber-400' : 'text-zinc-500'
                   }>
                     {usagePercent}%
                   </span>
                 </div>

                   <div
                     className={`h-1.5 rounded-full transition-all ${

               </div>

               {rl.throttle_count_24h > 0 && (
   ```

   Then add to Bindings:
   ```ts
   /** Rate limit binding: free tier (30 req/min) */
   RATE_LIMIT_FREE: RateLimitBinding
   /** Rate limit binding: pro tier (300 req/min) */
   RATE_LIMIT_PRO: RateLimitBinding
   /** Rate limit binding: enterprise tier (3000 req/min) */
   RATE_LIMIT_ENTERPRISE: RateLimitBinding
   /** Rate limit binding: IP-based pre-auth (100 req/10s) */
   RATE_LIMIT_IP: RateLimitBinding
   ```

3. In `apps/gateway/wrangler.toml`, add the rate limiting bindings using `[[unsafe.bindings]]` (the standard Cloudflare rate limiting config format):
   ```toml
   # Rate Limiting bindings (Cloudflare native, GA Sep 2025)
   # Per-key tier-based rate limits
   [[unsafe.bindings]]
   name = "RATE_LIMIT_FREE"
   type = "ratelimit"
   namespace_id = "0"
   simple = { limit = 30, period = 60 }

   [[unsafe.bindings]]
   name = "RATE_LIMIT_PRO"
   type = "ratelimit"
   namespace_id = "0"
   simple = { limit = 300, period = 60 }

   [[unsafe.bindings]]
   name = "RATE_LIMIT_ENTERPRISE"
   type = "ratelimit"
   namespace_id = "0"
   simple = { limit = 3000, period = 60 }

   # IP-based pre-auth defense (before API key validation)
   [[unsafe.bindings]]
   name = "RATE_LIMIT_IP"
   type = "ratelimit"
   namespace_id = "0"
   simple = { limit = 100, period = 10 }
   ```
   Also add a cron trigger section:
   ```toml

   [triggers]
   crons = ["0 3 * * *"]
   ```

   ```ts
   /**
    * D1 table schema for rate limit events.
    * Separate from usage table -- 429s are NOT metered as usage but tracked here.
    */
   export const RATE_LIMIT_EVENTS_TABLE_SCHEMA = `
   CREATE TABLE IF NOT EXISTS rate_limit_events (
     id INTEGER PRIMARY KEY AUTOINCREMENT,
     api_key_short TEXT NOT NULL,
     tier TEXT NOT NULL,
     ip TEXT NOT NULL,
     timestamp TEXT NOT NULL
   );

   CREATE INDEX IF NOT EXISTS idx_rle_api_key_short ON rate_limit_events(api_key_short);
   CREATE INDEX IF NOT EXISTS idx_rle_timestamp ON rate_limit_events(timestamp);
   `.trim()

   /**
    * Record a rate limit event to D1. Best-effort -- all errors silently caught.
    * Called via waitUntil when a 429 is returned.
    */
   export async function recordRateLimitEvent(
     db: D1Database,
     event: { api_key_short: string; tier: string; ip: string; timestamp: string }
   ): Promise<void> {
     try {
       await db
         .prepare(
           'INSERT INTO rate_limit_events (api_key_short, tier, ip, timestamp) VALUES (?, ?, ?, ?)'
         )
         .bind(event.api_key_short, event.tier, event.ip, event.timestamp)
         .run()
     } catch {
       // Best-effort: silently swallow
     }
   }
   ```

   try {
     result = await action.handler({
       params: actionParams,

       cursor,
     })
   } catch (handlerError) {
     const durationMs = Math.round(performance.now() - startTime)
     // Record failed request to usage table (error metering)
     c.executionCtx.waitUntil(
       recordUsage(c.env.USAGE_DB, {
         api_key_short: c.get('apiKeyRecord')?.shortToken ?? 'unknown',

         duration_ms: durationMs,

     )

1. Replace the stub in `apps/gateway/src/scheduled.ts` with actual retention logic:
   ```ts
   /**
    * Cron Trigger handler for data retention.
    * Runs daily at 3am UTC to clean up usage and rate limit event data older than 90 days.
    * All operations are best-effort -- failures are logged but don't propagate.
    */
   import type { AppEnv } from './lib/types'

   export async function handleScheduled(
     _event: ScheduledEvent,
     env: AppEnv['Bindings'],
     _ctx: ExecutionContext
   ): Promise<void> {
     const cutoff = new Date()
     cutoff.setDate(cutoff.getDate() - 90)
     const cutoffISO = cutoff.toISOString()

     try {
       // Delete old usage records
       const usageResult = await env.USAGE_DB
         .prepare('DELETE FROM usage WHERE timestamp < ?')
         .bind(cutoffISO)
         .run()
       console.log(`Retention: deleted ${usageResult.meta?.changes ?? 0} usage rows older than 90 days`)
     } catch (err) {
       console.error('Retention: failed to clean usage table:', err)
     }

     try {
       // Delete old rate limit events
       const rleResult = await env.USAGE_DB
         .prepare('DELETE FROM rate_limit_events WHERE timestamp < ?')
         .bind(cutoffISO)
         .run()
       console.log(`Retention: deleted ${rleResult.meta?.changes ?? 0} rate_limit_events rows older than 90 days`)
     } catch (err) {
       console.error('Retention: failed to clean rate_limit_events table:', err)
     }
   }
   ```

2. In `apps/gateway/src/routes/internal.ts`, add a GET /rate-limits endpoint:
   ```ts
   /**
    * GET /rate-limits
    *
    * Returns rate limit information per API key for dashboard display:
    * - Per key: tier, limit (req/min), usage count (last minute), throttle count (last 24h)
    * - Optionally filtered by ?key= short token
    */
   internal.get('/rate-limits', async (c) => {
     const keyFilter = c.req.query('key')

     try {
       // Get all keys with their tiers
       const keysResult = await c.env.AUTH_KV.list({ prefix: 'apikey:' })
       const keyData: Array<{
         short_token: string
         label: string | null
         tier: string
         limit: number
         usage_last_minute: number
         throttle_count_24h: number
       }> = []

       for (const kvKey of keysResult.keys) {
         const record = await c.env.AUTH_KV.get(kvKey.name, 'json') as any
         if (!record) continue
         if (keyFilter && record.shortToken !== keyFilter) continue

         const tier = record.tier ?? 'free'
         const tierLimits: Record<string, number> = { free: 30, pro: 300, enterprise: 3000 }
         const limit = tierLimits[tier] ?? 30

         // Get usage in last minute
         let usageLastMinute = 0
         try {
           const usageResult = await c.env.USAGE_DB
             .prepare(
               `SELECT COUNT(*) as count FROM usage
                WHERE api_key_short = ? AND timestamp >= datetime('now', '-1 minute')`
             )
             .bind(record.shortToken)
             .first<{ count: number }>()
           usageLastMinute = usageResult?.count ?? 0
         } catch {
           // Table may not exist
         }

         // Get throttle count in last 24h
         let throttleCount24h = 0
         try {
           const throttleResult = await c.env.USAGE_DB
             .prepare(
               `SELECT COUNT(*) as count FROM rate_limit_events
                WHERE api_key_short = ? AND timestamp >= datetime('now', '-24 hours')`
             )
             .bind(record.shortToken)
             .first<{ count: number }>()
           throttleCount24h = throttleResult?.count ?? 0
         } catch {
           // Table may not exist
         }

         keyData.push({
           short_token: record.shortToken,
           label: record.label ?? null,
           tier,
           limit,
           usage_last_minute: usageLastMinute,
           throttle_count_24h: throttleCount24h,
         })
       }

       return c.json({
         ok: true,
         data: keyData,
       })
     } catch {
       return c.json({
         ok: true,
         data: [],
       })
     }
   })
   ```

   Import TIER_LIMITS from auth/types at the top of internal.ts if needed (or use inline constant as shown above for simplicity -- either approach works, the inline approach avoids an extra import from the auth module).

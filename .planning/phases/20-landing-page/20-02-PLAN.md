Create `apps/dashboard/src/components/landing-redirect.tsx` — a 'use client' component that extracts the current page.tsx redirect logic. This component renders nothing visible (returns null). On mount, it checks:
1. If `isDemo` (from `useDemo()`) is true, redirect to `/overview`
2. If `getAdminToken()` returns a token, redirect to `/overview`
3. Otherwise, do nothing (user stays on landing page)

Note: The current page.tsx redirects unauthenticated users to `/login`. With the landing page, unauthenticated users should STAY on the landing page (they can click the CTA to go to /login). So remove the `else { router.replace('/login') }` branch. Only redirect authenticated/demo users away.

```tsx
'use client';

import { useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { getAdminToken } from '@/lib/auth';
import { useDemo } from '@/lib/demo-context';

export function LandingRedirect() {
  const router = useRouter();
  const { isDemo } = useDemo();

  useEffect(() => {
    if (isDemo) {
      router.replace('/overview');
      return;
    }
    const token = getAdminToken();
    if (token) {
      router.replace('/overview');
    }
  }, [router, isDemo]);

  return null;
}
```

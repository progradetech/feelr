  skipped: boolean

  duration_ms: number

1. Create `apps/gateway/src/lib/chain-executor.ts`:

```ts
import type { ChainDefinition, ChainExecutionResult, ChainStepResult, ResolveContext } from './chain-types'
import { resolveTemplate, evalCondition, validateChain, MAX_STEPS } from './chain-types'
import { getConnector } from '../connectors/registry'
import { getCredential } from '../auth/credentials'
import { FeelrError } from './errors'

interface ExecuteChainOpts {
  chain: ChainDefinition
  params: Record<string, string>
  env: {
    AUTH_KV: KVNamespace
    ENCRYPTION_KEY: string
  }
}

export async function executeChain(opts: ExecuteChainOpts): Promise<ChainExecutionResult>
```

**Implementation:**
1. Validate chain definition via validateChain(). Throw FeelrError on invalid.
2. Validate required params are provided (check against chain.params where required=true). Apply defaults for missing optional params.
3. Initialize ResolveContext with params and empty steps.
4. For each step:
   a. Evaluate `if` condition -> skip if false (set step output to null)
   b. Resolve all `with` values using resolveTemplate
   c. Parse `uses` -> connector name + action name (split on `/`)
   d. Look up connector via getConnector(connectorName) -> throw if not found
   e. Look up action via connector.actions[actionName] -> throw if not found
   f. Get credential via getCredential(connectorName, env.AUTH_KV, env.ENCRYPTION_KEY)
   g. Call action.handler({ params: resolvedParams, fetch, credential }) directly
   h. Handle retry: wrap in try/catch loop for retry.max_attempts
   i. Store result data in ctx.steps[step.id]
   j. Record StepResult
5. Return ChainExecutionResult with all step results and final step output.

Key difference from Go executor: this calls `action.handler()` DIRECTLY from the connector registry -- no HTTP round-trip. Credential fetched per-step (different connectors may need different credentials).

2. Create `apps/gateway/src/routes/chains.ts`:

```ts
import { OpenAPIHono } from '@hono/zod-openapi'
import type { AppEnv } from '../lib/types'
import { wrapResponse } from '../lib/envelope'
import { FeelrError } from '../lib/errors'
import { executeChain } from '../lib/chain-executor'
import type { ChainDefinition } from '../lib/chain-types'

const chains = new OpenAPIHono<AppEnv>()

/**
 * POST /v1/chains/run
 * Execute a chain definition server-side.
 * Body: { chain: ChainDefinition, params: Record<string, string> }
 */
chains.post('/run', async (c) => {
  const body = await c.req.json()
  const chainDef = body.chain as ChainDefinition
  const params = (body.params || {}) as Record<string, string>

  if (!chainDef || !chainDef.name || !chainDef.steps) {
    throw new FeelrError('INVALID_CHAIN', {
      message: 'Request body must include a valid chain definition with name and steps',
      hint: 'abort',
      status: 400,
    })
  }

  const result = await executeChain({
    chain: chainDef,
    params,
    env: {
      AUTH_KV: c.env.AUTH_KV,
      ENCRYPTION_KEY: c.env.ENCRYPTION_KEY,
    },
  })

  const requestId = c.get('requestId')
  return c.json(wrapResponse({
    data: {
      success: result.success,
      steps: result.steps,
      output: result.final_output,
    },
    meta: {
      request_id: requestId,
      connector: 'chain',
      action: chainDef.name,
      duration_ms: result.steps.reduce((sum, s) => sum + s.duration_ms, 0),
    },
  }))
})

export const chainsRoutes = chains
```

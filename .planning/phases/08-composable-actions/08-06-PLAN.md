1. **github-slack-issue-notify.yaml** (GitHub + Slack):
```yaml
name: github-slack-issue-notify
description: "Creates a GitHub issue and notifies a Slack channel with the issue link"
version: "1.0"

params:
  - name: repo
    type: string
    required: true
    description: "Repository in owner/repo format"
  - name: title
    type: string
    required: true
    description: "Issue title"
  - name: body
    type: string
    required: false
    description: "Issue body text"
    default: ""
  - name: channel
    type: string
    required: true
    description: "Slack channel to notify (e.g., #engineering)"

retry:
  max_attempts: 2
  delay_seconds: 3

steps:
  - id: create_issue
    uses: github/issues.create
    with:
      repo: "${{ params.repo }}"
      title: "${{ params.title }}"
      body: "${{ params.body }}"

  - id: notify_slack
    uses: slack/messages.send
    if: "${{ steps.create_issue.number }} > 0"
    with:
      channel: "${{ params.channel }}"
      text: "New issue #${{ steps.create_issue.number }} created in ${{ params.repo }}: ${{ params.title }} - ${{ steps.create_issue.html_url ?? 'no link' }}"
```

1. **stripe-slack-payment-alert.yaml** (Stripe + Slack):
```yaml
name: stripe-slack-payment-alert
description: "Checks recent Stripe payments and alerts Slack about the latest payment"
version: "1.0"

params:
  - name: channel
    type: string
    required: true
    description: "Slack channel for payment alerts"
  - name: limit
    type: string
    required: false
    description: "Number of recent payments to check"
    default: "1"

retry:
  max_attempts: 2
  delay_seconds: 3

steps:
  - id: check_payments
    uses: stripe/payments.list
    with:
      limit: "${{ params.limit }}"

  - id: alert_slack
    uses: slack/messages.send
    if: "exists ${{ steps.check_payments.id ?? '' }}"
    with:
      channel: "${{ params.channel }}"
      text: "Latest Stripe payment: ${{ steps.check_payments.amount ?? '0' }} ${{ steps.check_payments.currency ?? 'usd' }} — Status: ${{ steps.check_payments.status ?? 'unknown' }}"
```

2. **github-stripe-customer-issue.yaml** (GitHub + Stripe):
```yaml
name: github-stripe-customer-issue
description: "Looks up a Stripe customer and creates a GitHub issue with their details"
version: "1.0"

params:
  - name: customer_id
    type: string
    required: true
    description: "Stripe customer ID (cus_xxx)"
  - name: repo
    type: string
    required: true
    description: "Repository in owner/repo format for the issue"

retry:
  max_attempts: 2
  delay_seconds: 3

steps:
  - id: get_customer
    uses: stripe/customers.get
    with:
      customer_id: "${{ params.customer_id }}"

  - id: create_issue
    uses: github/issues.create
    with:
      repo: "${{ params.repo }}"
      title: "Customer inquiry: ${{ steps.get_customer.name ?? steps.get_customer.email ?? 'Unknown' }}"
      body: "Customer: ${{ steps.get_customer.name ?? 'N/A' }}\nEmail: ${{ steps.get_customer.email ?? 'N/A' }}\nStripe ID: ${{ params.customer_id }}"
```

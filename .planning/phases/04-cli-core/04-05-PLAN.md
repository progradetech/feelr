**Dynamic completion for `run` command (update cli/cmd/run.go):**

Add `ValidArgsFunction` to the run command:
```go
ValidArgsFunction: func(cmd *cobra.Command, args []string, toComplete string) ([]string, cobra.ShellCompDirective) {
    // Need config and client for gateway calls
    cfg, err := config.Load(getProfile(cmd))
    if err != nil || cfg.APIKey == "" {
        return nil, cobra.ShellCompDirectiveNoFileComp
    }
    client := client.NewGatewayClient(cfg.Gateway, cfg.APIKey)

    if len(args) == 0 {
        // Complete connector names
        resp, err := client.GetTools("")
        if err != nil {
            return nil, cobra.ShellCompDirectiveNoFileComp
        }
        // Parse connector names from response data
        var connectors []struct{ Name string `json:"name"` }
        json.Unmarshal(resp.Data, &connectors)
        names := make([]string, len(connectors))
        for i, c := range connectors {
            names[i] = c.Name
        }
        return names, cobra.ShellCompDirectiveNoFileComp
    }
    if len(args) == 1 {
        // Complete action names for the given connector
        resp, err := client.GetTools("/" + args[0])
        if err != nil {
            return nil, cobra.ShellCompDirectiveNoFileComp
        }
        var connector struct {
            Actions []struct{ Name string `json:"name"` } `json:"actions"`
        }
        json.Unmarshal(resp.Data, &connector)
        names := make([]string, len(connector.Actions))
        for i, a := range connector.Actions {
            names[i] = a.Name
        }
        return names, cobra.ShellCompDirectiveNoFileComp
    }
    // No completion for key=value params
    return nil, cobra.ShellCompDirectiveNoFileComp
}
```
